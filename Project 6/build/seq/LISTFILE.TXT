    3 /*  Project 6 - sequential reference for MeshIntegrate.c.                   */
    4 /*  One process evaluates all N trapezoids with exactly the same            */
    5 /*  arithmetic, giving the T(1) baseline and the reference answer.          */
    6 
    7 #include <math.h>
    8 
    9 #define N        28800      /* total number of trapezoids                  */
   10 
   11 float f(float t) {
   12   return( sqrt(4 - t*t) );
   13 }
   14 
   15 main() {
   16   float a, b, w, t, sum, err;
   17   int j;
   18 
   19   a = 0.0;
   20   b = 2.0;
   21   w = (b - a) / N;
   22   sum = 0.0;
   23   t = a;
   24   for (j = 1; j <= N; j++) {
   25     sum = sum + f(t);
   26     t = t + w;
   27   }
   28   sum = w * sum + w / 2 * (f(b) - f(a));
   29 
   30   err = sum - 3.14159265358979;
   31   if (err < 0) err = -err;
   32   cout << "=======================================================" << endl;
   33   cout << " NUMERICAL INTEGRATION - sequential reference" << endl;
   34   cout << "=======================================================" << endl;
   35   cout << " f(t) = sqrt(4 - t*t),  a = 0,  b = 2" << endl;
   36   cout << " Trapezoids N         = " << N << endl;
   37   cout.precision(12);
   38   cout << endl << " Integral             = " << sum << endl;
   39   cout << " pi                   = " << 3.14159265358979 << endl;
   40   cout << " Absolute error       = " << err << endl;
   41   cout.precision(9);

   43   cout << "=======================================================" << endl;
   44 }
